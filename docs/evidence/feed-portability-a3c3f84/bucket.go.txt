package main
import("context";"fmt";"time";"github.com/aws/aws-sdk-go-v2/aws";"github.com/aws/aws-sdk-go-v2/config";"github.com/aws/aws-sdk-go-v2/credentials";"github.com/aws/aws-sdk-go-v2/service/s3")
func main(){ctx,cancel:=context.WithTimeout(context.Background(),15*time.Second);defer cancel();c,err:=config.LoadDefaultConfig(ctx,config.WithRegion("us-east-1"),config.WithCredentialsProvider(credentials.NewStaticCredentialsProvider("portability-local","portability-local-only-password","")));if err!=nil{panic(err)};s:=s3.NewFromConfig(c,func(o *s3.Options){o.BaseEndpoint=aws.String("http://127.0.0.1:59005");o.UsePathStyle=true});_,err=s.CreateBucket(ctx,&s3.CreateBucketInput{Bucket:aws.String("feed-portability-synthetic")});if err!=nil{panic(err)};v,err:=s.GetBucketVersioning(ctx,&s3.GetBucketVersioningInput{Bucket:aws.String("feed-portability-synthetic")});if err!=nil||v.Status!=""{panic(fmt.Sprintf("bucket version check: %v %v",v,err))};fmt.Println("created dedicated unversioned synthetic bucket")}
